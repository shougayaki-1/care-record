function ShiftVarianceDemo() {
  const [expanded, setExpanded] = useState<Record<string, boolean>>({ 'shift-2': true });
  const totals = rows.reduce((acc, row) => ({
    planned: acc.planned + (row.plannedH ?? 0),
    actual: acc.actual + (row.actualH ?? 0),
  }), { planned: 0, actual: 0 });
  const diff = totals.actual - totals.planned;

  return (
    <ManualDemoFrame title="シフト差異合計と詳細表示">
      <ManualScreenHighlight number={1} label="サマリーで全体差異を把握">
        <Paper variant="outlined" sx={{ overflow: 'hidden' }}>
          <Box sx={{ px: { xs: 1.5, sm: 2 }, py: 1.5, bgcolor: 'background.tint', borderBottom: 1, borderColor: 'divider' }}>
            <Stack direction={{ xs: 'column', sm: 'row' }} spacing={1.5} justifyContent="space-between" alignItems={{ xs: 'stretch', sm: 'center' }}>
              <Typography variant="subtitle2" fontWeight={800}>シフト差異合計</Typography>
              <Stack direction="row" spacing={1} useFlexGap flexWrap="wrap">
                <Chip size="small" variant="outlined" label={`予定 ${totals.planned.toFixed(1)}h`} />
                <Chip size="small" color="primary" label={`実績 ${totals.actual.toFixed(1)}h`} />
                <Chip size="small" color="warning" variant="outlined" label={`差異 ${diff >= 0 ? '+' : ''}${diff.toFixed(1)}h`} />
              </Stack>
            </Stack>
          </Box>

          <ShiftVarianceTable>
            <TableContainer sx={{ overflowX: 'auto' }}>
              <Table size="small">
                <TableHead sx={{ bgcolor: 'background.tint' }}>
                  <TableRow>
                    <TableCell />
                    <TableCell sx={{ fontWeight: 800 }}>シフト日時</TableCell>
                    <TableCell sx={{ fontWeight: 800 }}>利用者名</TableCell>
                    <TableCell sx={{ fontWeight: 800 }}>担当職員</TableCell>
                    <TableCell align="right" sx={{ fontWeight: 800 }}>予定(h)</TableCell>
                    <TableCell align="right" sx={{ fontWeight: 800 }}>実績(h)</TableCell>
                    <TableCell align="right" sx={{ fontWeight: 800 }}>差異(h)</TableCell>
                    <TableCell sx={{ fontWeight: 800 }}>操作</TableCell>
                  </TableRow>
                </TableHead>
                <TableBody>
                  {rows.map((row) => {
                    const open = Boolean(expanded[row.id]);
                    return (
                      <TableRow key={row.id} hover>
                        <TableCell padding="checkbox" sx={{ verticalAlign: 'top' }}>
                          <IconButton size="small" onClick={() => setExpanded((current) => ({ ...current, [row.id]: !open }))}>
                            {open ? <ExpandLessIcon fontSize="small" /> : <ExpandMoreIcon fontSize="small" />}
                          </IconButton>
                        </TableCell>
                        <TableCell sx={{ verticalAlign: 'top' }}>{row.start} - {row.end}</TableCell>
                        <TableCell sx={{ verticalAlign: 'top' }}>{row.clientName}</TableCell>
                        <TableCell sx={{ verticalAlign: 'top' }}>
                          <Stack spacing={0.75}>
                            <Typography variant="body2">{row.staffNames}</Typography>
                            <Stack direction="row" spacing={0.5} useFlexGap flexWrap="wrap">
                              {row.flags?.map((flag) => (
                                <Chip key={flag} label={flag === '担当不一致' && row.actualStaffNames ? `担当不一致 実績: ${row.actualStaffNames}` : flag} size="small" color="warning" variant="outlined" />
                              ))}
                            </Stack>
                          </Stack>
                        </TableCell>
                        <TableCell align="right" sx={{ verticalAlign: 'top' }}>{row.plannedH?.toFixed(1) ?? '-'}</TableCell>
                        <TableCell align="right" sx={{ verticalAlign: 'top' }}>{row.actualH?.toFixed(1) ?? '-'}</TableCell>
                        <TableCell align="right" sx={{ verticalAlign: 'top', color: row.diffH != null && row.diffH < -0.1 ? 'error.main' : 'text.primary', fontWeight: row.diffH != null && row.diffH < -0.1 ? 800 : 400 }}>
                          {row.diffH == null ? '-' : `${row.diffH >= 0 ? '+' : ''}${row.diffH.toFixed(1)}`}
                        </TableCell>
                        <TableCell sx={{ verticalAlign: 'top' }}>
                          <Stack direction="row" spacing={0.75} useFlexGap flexWrap="wrap">
                            {row.actualH == null ? (
                              <Button size="small" variant="contained" startIcon={<EditNoteIcon />} onClick={() => undefined}>実績作成</Button>
                            ) : (
                              <Button size="small" variant="outlined" startIcon={<EditNoteIcon />} onClick={() => undefined}>実績確認</Button>
                            )}
                            <Button size="small" variant="outlined" color="inherit" startIcon={<EventNoteIcon />} onClick={() => undefined}>シフト確認</Button>
                          </Stack>
                          <Collapse in={open} timeout="auto" unmountOnExit>
                            <Divider sx={{ my: 1.5 }} />
                            <Stack spacing={1}>
                              {row.details.map((detail) => (
                                <Box key={`${row.id}-${detail.kind}`}>
                                  <Stack direction="row" justifyContent="space-between" alignItems="center">
                                    <Chip size="small" label={detail.label} color={detail.kind === 'actual' ? 'primary' : 'default'} variant={detail.kind === 'actual' ? 'filled' : 'outlined'} />
                                    <Typography variant="body2" fontWeight={800}>{detail.hours.toFixed(2)}h</Typography>
                                  </Stack>
                                  <Typography variant="body2" color="text.secondary" sx={{ mt: 0.5 }}>{detail.note}</Typography>
                                </Box>
                              ))}
                            </Stack>
                          </Collapse>
                        </TableCell>
                      </TableRow>
                    );
                  })}
                </TableBody>
              </Table>
            </TableContainer>
          </ShiftVarianceTable>
        </Paper>
      </ManualScreenHighlight>
      <ManualCallout title="運用で見る場所">
        月末締め前に、差異が大きい行、実績なし、担当不一致、サービス不一致の順に確認すると、帳票や請求前の手戻りを減らせます。
      </ManualCallout>
    </ManualDemoFrame>
  );
}
